🎯 Objetivo

Criar um programa que:

  1. Tenha um saldo inicial
  2. Mostre um menu com opções:
    • Ver saldo
    • Depositar
    • Sacar
    • Sair
  3. Não permita sacar mais do que tem no saldo
  4. Continue rodando até o usuário escolher sair

function caixaEletronico() { let saldo = 1000; let continuar = true; while (continuar) { let opcao = prompt( "Escolha uma opção:\n" + "1 - Ver saldo\n" + "2 - Depositar\n" + "3 - Sacar\n" + "4 - Sair" ); if (opcao === "1") { console.log("Seu saldo é: R$ " + saldo); } else if (opcao === "2") { let deposito = parseFloat(prompt("Digite o valor para depositar:")); saldo = saldo + deposito; console.log("Depósito realizado. Novo saldo: R$ " + saldo); } else if (opcao === "3") { let saque = parseFloat(prompt("Digite o valor para sacar:")); if (saque <= saldo) { saldo = saldo - saque; console.log("Saque realizado. Novo saldo: R$ " + saldo); } else { console.log("Saldo insuficiente!"); } } else if (opcao === "4") { continuar = false; console.log("Obrigado por usar o caixa eletrônico."); } else { console.log("Opção inválida!"); } } } caixaEletronico();
